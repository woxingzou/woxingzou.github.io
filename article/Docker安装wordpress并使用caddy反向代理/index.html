compose.yaml配置

services:
  mysql:
    image: mysql:8.0.40
    container_name: mysql
    restart: unless-stopped
    user: ${UID}:${GID}
    networks:
      - docker_net0
    ports:
      - "3306:3306"
    environment:
      TZ: Asia/Shanghai
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:-admin@123456}
      MYSQL_DATABASE: ${MYSQL_DATABASE:-wordpress}
    command:
      # 将mysql8.0默认密码策略 修改为 原先 策略 (mysql8.0对其默认策略做了更改 会导致密码无法匹配)
      --default-authentication-plugin=mysql_native_password
      --explicit_defaults_for_timestamp=true
    volumes:
      # 数据挂载
      - ${HOME}/data/mysql/data:/var/lib/mysql
      # 配置挂载
      - ${HOME}/data/mysql/conf/my.cnf:/etc/my.cnf

  caddy:
    image: caddy:2.8.4-alpine
    container_name: caddy
    restart: unless-stopped
    user: ${UID}:${GID}
    networks:
      - docker_net0
    ports:
      - "80:80"
      - "443:443"
      - "443:443/udp"
    volumes:
      - ${HOME}/data/caddy/Caddyfile:/etc/caddy/Caddyfile
      - ${HOME}/data/caddy/sites:/etc/caddy/sites
      - ${HOME}/data/caddy/srv:/srv
      - ${HOME}/data/caddy/data:/data
      - ${HOME}/data/caddy/config:/config

  wordpress:
    build:
      context: wordpress
      dockerfile: Dockerfile
    image: wordpress:latest
    container_name: wordpress
    restart: unless-stopped
    user: ${UID}:${GID}
    depends_on:
      - mysql
      - caddy
    networks:
      - docker_net0
    ports:
      - "8080:80"
    environment:
      TZ: Asia/Shanghai
      WORDPRESS_DB_HOST: mysql
      WORDPRESS_DB_USER: root
      WORDPRESS_DB_PASSWORD: admin@123456
      WORDPRESS_DB_NAME: wordpress
    volumes:
      - ${HOME}/data/wordpress:/var/www/html

networks:
  docker_net0:

安装脚本install.sh

#!/bin/bash

DATA_DIR=$HOME/data
DOCKER_DIR=$HOME/docker

CADDY_DIR=$DATA_DIR/caddy
MYSQL_DIR=$DATA_DIR/mysql
WORDPRESS_DIR=$DATA_DIR/wordpress

echo -e "UID=$(id -u)\nGID=$(id -g)" > $DOCKER_DIR/.env

mkdir -p $CADDY_DIR/{data,config,srv,sites} \
         $MYSQL_DIR/{conf,data} \
         $WORDPRESS_DIR

cp $DOCKER_DIR/caddy/Caddyfile $CADDY_DIR/Caddyfile
cp -r $DOCKER_DIR/mysql/conf $MYSQL_DIR

docker compose up --build -d mysql caddy wordpress

执行安装

cd $HOME/docker && bash install.sh

验证安装

docker ps -a

caddy反向代理

创建并编辑$HOME/data/caddy/sites/your.domain.com.conf

your.domain.com { # 替换成你的域名
	encode zstd gzip

	request_body {
		max_size 100MB
	}

	reverse_proxy wordpress:80
}

注意

注意替换你的域名

登录wordpress后台网站进行配置

浏览器登陆你的网站https://your.domain.com

应该只需要设置你的用户名和密码，其他什么都不需要做
